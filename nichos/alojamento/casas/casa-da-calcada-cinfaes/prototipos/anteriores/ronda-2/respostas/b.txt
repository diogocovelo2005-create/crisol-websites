    <script>
        // Check for reduced motion preference
        const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

        if (!prefersReducedMotion) {
            // Setup Lenis for smooth scrolling
            const lenis = new Lenis({
                duration: 1.2,
                easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t))
            });

            function raf(time) {
                lenis.raf(time);
                requestAnimationFrame(raf);
            }
            requestAnimationFrame(raf);

            // Connect Lenis to ScrollTrigger
            lenis.on('scroll', ScrollTrigger.update);
            gsap.ticker.add((time) => { lenis.raf(time * 1000); });
            gsap.ticker.lagSmoothing(0);

            gsap.registerPlugin(ScrollTrigger);

            // 1. Text reveals
            // Set initial state via CSS class .text-anim (opacity 0, y 30)
            const textElements = gsap.utils.toArray('.text-anim');
            
            // For the hero title, animate immediately
            gsap.to('.hero-title', { opacity: 1, y: 0, duration: 1.5, ease: "power3.out" });

            textElements.forEach(el => {
                if(!el.classList.contains('hero-title')) {
                    gsap.to(el, {
                        opacity: 1,
                        y: 0,
                        duration: 1.2,
                        ease: "power3.out",
                        scrollTrigger: {
                            trigger: el,
                            start: "top 90%",
                            toggleActions: "play none none none"
                        }
                    });
                }
            });

            // 2. Image expansion mask effect
            const imgWrappers = gsap.utils.toArray('.rev-img');
            
            imgWrappers.forEach(wrapper => {
                const img = wrapper.querySelector('img');
                
                // Initialize scale
                gsap.set(img, { scale: 1.2 });

                gsap.to(img, {
                    scale: 1,
                    ease: "none",
                    scrollTrigger: {
                        trigger: wrapper,
                        start: "top 95%",
                        end: "bottom 20%",
                        scrub: true
                    }
                });
            });
        } else {
            // If reduced motion, remove the init classes so content shows normally
            document.documentElement.classList.add('no-js');
        }
    </script>
