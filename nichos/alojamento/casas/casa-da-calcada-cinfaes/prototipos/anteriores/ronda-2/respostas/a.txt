        const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');

        if (!mediaQuery.matches) {
            // Smooth scroll setup
            const lenis = new Lenis({
                duration: 1.2,
                easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
                direction: 'vertical',
                gestureDirection: 'vertical',
                smooth: true,
                mouseMultiplier: 1,
                smoothTouch: false,
                touchMultiplier: 2,
            });

            function raf(time) {
                lenis.raf(time);
                requestAnimationFrame(raf);
            }
            requestAnimationFrame(raf);

            // Parallax Effects for backgrounds
            gsap.utils.toArray('.parallax-bg').forEach(bg => {
                const speed = bg.dataset.speed || 0.5;
                gsap.to(bg, {
                    yPercent: 20 * speed,
                    ease: "none",
                    scrollTrigger: {
                        trigger: bg.parentElement,
                        start: "top bottom",
                        end: "bottom top",
                        scrub: true
                    }
                });
            });

            // Float/Slide effects for polaroids
            gsap.utils.toArray('.polaroid').forEach(polaroid => {
                const speed = polaroid.dataset.speed || 1;
                gsap.fromTo(polaroid, 
                    { y: 100 },
                    {
                        y: -50 * speed,
                        ease: "none",
                        scrollTrigger: {
                            trigger: polaroid.parentElement,
                            start: "top bottom",
                            end: "bottom top",
                            scrub: true
                        }
                    }
                );
            });
        }
        
        // Hide elements with specific mobile classes via JS if needed, but CSS handles it via .hidden-mobile in reduced screen sizes if we define it.
        // Actually, let's just let CSS media queries handle mobile layout shifts completely.
    </script>
